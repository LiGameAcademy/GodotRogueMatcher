### 2. 创建 LevelUpPopup UI

```gdscript
# ui/level_up_popup.gd
extends UIPopup
class_name LevelUpPopup

signal item_selected(item: ItemData)

@onready var option_1: ItemCard = $Options/Option1
@onready var option_2: ItemCard = $Options/Option2
@onready var option_3: ItemCard = $Options/Option3

func show_options(items: Array[ItemData]) -> void:
    # 显示三个选项
    option_1.set_item(items[0])
    option_2.set_item(items[1])
    option_3.set_item(items[2])
    
    # 显示界面
    visible = true
    get_tree().paused = true

func _on_option_selected(item: ItemData) -> void:
    item_selected.emit(item)
    visible = false
    get_tree().paused = false
```

### 3. 道具自动放置

```gdscript
# scripts/global/item_placer.gd
extends Node
class_name ItemPlacer

## 自动放置道具到随机位置
func place_item_randomly(board: Board, item: ItemData) -> void:
    var empty_cells = board.get_empty_cells()
    if empty_cells.is_empty():
        print("警告：没有空位放置道具")
        return
    
    # 随机选择空位
    var random_index = randi() % empty_cells.size()
    var cell = empty_cells[random_index]
    
    # 放置道具
    place_item_at(board, cell, item)

## 在指定位置放置道具
func place_item_at(board: Board, cell: Cell, item: ItemData) -> void:
    var item_node = item.scene.instantiate()
    cell.add_child(item_node)
    
    # 播放出现动画
    play_spawn_animation(item_node)
```

### 2. 在 main.gd 中处理升级

```gdscript

    # 连接升级信号
```

## 四、UI 设计

### 1. 界面布局

- **全屏遮罩：** 半透明黑色背景
- **中央面板：** 显示三个道具选项
- **标题：** "LEVEL UP!" 或 "等级提升！"
- **道具卡片：** 3 个并排显示

### 2. 道具卡片

- **图标：** 64x64 像素
- **名称：** 道具名称
- **描述：** 简短效果说明
- **稀有度边框：** 不同颜色标识

## 五、优势总结

1. ✅ **简化开发：** 无需商店、金币系统
2. ✅ **节奏感强：** 定期升级保持兴奋
3. ✅ **策略深度：** 三选一增加决策乐趣
4. ✅ **易于扩展：** 可以轻松添加新道具
5. ✅ **符合肉鸽：** 类似 Survivors 的体验
